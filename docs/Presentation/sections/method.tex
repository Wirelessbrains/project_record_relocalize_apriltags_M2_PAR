\begin{outlinecard}[Main Equation]
  \scriptsize
  \only<1>{
    \textbf{Recording}
    \[
      {}^{\mathrm{cam}}T_{\mathrm{tag}}(t)
    \]
  }
  \only<2>{
    \textbf{Build + Optimize}
    \[
      \min_{\{{}^{w}T_{\mathrm{tag}_i}\}}
      \sum_{(i,j)\in\mathcal{E}}
      \rho\!\left(\|\mathrm{err}_{ij}\|^2\right)
    \]
  }
  \only<3->{
    \textbf{Relocalization}
    \[
      D=\|\mathbf{p}_{\mathrm{cur}}-\mathbf{p}_{\mathrm{nearest}}\|
    \]
    \[
      \phi=\mathrm{wrap}(\psi_{\mathrm{cur}}-\psi_{\mathrm{nearest}})
    \]
  }
\end{outlinecard}
